\newcommand{\figPhaseTransition}[1]{
  \begin{figure}[htbp]
    \centering
    \caption{Fases de transición de estados de una sustancia} 
    \label{#1}
    \vspace{6pt}

    \begin{tikzpicture}[
      node distance=3cm,
      every node/.style={font=\small},
      % Estilo para simplificar las flechas con offset y rotación
      trans/.style={arrow_yes, sloped, midway} 
      ]

      % Nodos principales
      \node (gas) at (5,8.66) {Gas};
      \node (solid) at (0,2) {Sólido};
      \node (liquid) at (10,2) {Líquido};

      % Flechas Sólido <-> Líquido
      \draw[trans] ($(solid.east)+(0,0.1)$) -- node[above]{Fusión} ($(liquid.west)+(0,0.1)$);
      \draw[trans] ($(liquid.west)-(0,0.1)$) -- node[below]{Solidificación} ($(solid.east)-(0,0.1)$);

      % Flechas Líquido <-> Gas
      \draw[trans] ($(liquid.north west)+(0.1,0.1)$) -- node[above]{Vaporización} ($(gas.south east)+(0.1,0.1)$);
      \draw[trans] ($(gas.south east)-(0.1,0.1)$) -- node[below]{Condensación} ($(liquid.north west)-(0.1,0.1)$);

      % Flechas Sólido <-> Gas
      \draw[trans] ($(solid.north east)+(-0.1,0.1)$) -- node[above]{Sublimación} ($(gas.south west)+(-0.1,0.1)$);
      \draw[trans] ($(gas.south west)+(0.1,-0.1)$) -- node[below]{Deposición o sublimación inversa} ($(solid.north east)+(0.1,-0.1)$);

    \end{tikzpicture}
    
    \begin{flushleft}
        \small \textit{Nota.} Elaborado con el paquete TikZ.
    \end{flushleft}
\end{figure}
}
