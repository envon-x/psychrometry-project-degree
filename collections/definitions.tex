%%%% Definiciones de variables termodinámicas 
%%%% Para representar la constante de gases
\def\idealConstantGas{\ensuremath{R}}

%%%% Componente Líquido
\def\liquid{\ensuremath{{l}}}

%%%% Componente vapor
\def\vapor{\ensuremath{{v}}}

%%%% Símbolo para la representación de molar
\def\molar{\ensuremath{{m}}}

%%%% Componente gas
\def\gas{\ensuremath{{g}}} % 

% Para las propiedades críticas
\newcommand{\critical}{{{c}}}

% Para las propiedades del punto triple
\newcommand{\triple}{{{t}}}

% Sufijo para las las variables (usado en presión, calor)
% Indica la suma total de los valores de las variables
\def\total{\ensuremath{{tot}}}

% Usado para el calor latente
\def\latent{\ensuremath{{L}}}

% Este símbolo se usa para representar la saturación
\newcommand{\saturated}{{{o}}}

% Símbolo que usar para la representación de las condiciones atmosféricas
\newcommand{\atmospheric}{{{atm}}}

% Símbolo Usado para el calor sensible
\def\sensible{\ensuremath{{s}}}

%%%%%%%%%%%%%%%% Psychrometric variables %%%%%%%%%%%%%%%%
%%%% Temperatura de bulbo seco
\def\dryBulbSymbol{\ensuremath{{bs}}}

%%%% Temperatura de bulbo húmedo
\def\wetBulbSymbol{\ensuremath{{bh}}}

%%%% Temperatura de rocío,  temperatura a la que, durante un proceso
%%%% de enfriamiento, empieza a condensarse el vapor de agua contenido 
%%%% en el aire, produciendo rocío, neblina, cualquier tipo de nube 
%%%% o, en caso de que la temperatura sea lo suficientemente baja, escarcha.
\def\dewPointSymbol{\ensuremath{{roc}}}

%%%%%%%%%%%%%%%% UNIDADES PSICROMÉTRICAS %%%%%%%%%%%%%%%%

%%%% Humedad específica: g/kg
\def\specificHumidityUnit{\ensuremath{{{\unit{g}}/{\unit{kg}}}}}
