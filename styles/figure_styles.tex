% #2: Título (Caption)

\newcommand{\figuraAPA}[4]{
    \begin{figure}[H]
        \centering
        \caption{#2}
        \label{#1}
        \input{#3}
        \vspace{6pt}
        \begin{flushleft}
            \small \textit{Nota.} #4
        \end{flushleft}
    \end{figure}
}
